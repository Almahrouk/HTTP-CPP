Process Management

Process
⚙
execve
int execve(const char *path, char *const argv[], char *const envp[])

Replaces the current process image with a new program. The calling process is completely replaced — there is no return on success. argv[] provides arguments and envp[] provides environment variables. It does not create a new process; it transforms the current one.

returns -1 on error errno: EACCES, ENOENT, E2BIG #include <unistd.h>
Example
#include <unistd.h>
#include <stdio.h>

int main() {
    char *args[] = { "/bin/ls", "-la", "/tmp", NULL };
    char *env[]  = { "HOME=/root", NULL };

    execve("/bin/ls", args, env);

    /* Only reached if execve fails */
    perror("execve");
    return 1;
}
🍴
fork
pid_t fork(void)

Creates a new child process as an exact copy of the calling process. Both processes continue from the point after the call. Returns 0 to the child and the child's PID to the parent. Returns -1 on failure.

0 → child, PID → parent errno: EAGAIN, ENOMEM #include <unistd.h>
Example
#include <unistd.h>
#include <stdio.h>

int main() {
    pid_t pid = fork();

    if (pid < 0)       perror("fork");
    else if (pid == 0)  printf("Child: PID=%d\n", getpid());
    else               printf("Parent: child PID=%d\n", pid);

    return 0;
}
⏳
waitpid
pid_t waitpid(pid_t pid, int *status, int options)

Suspends the calling process until a child specified by pid changes state. Use WIFEXITED and WEXITSTATUS macros to inspect status. Pass WNOHANG for non-blocking check.

returns child PID or -1 errno: ECHILD, EINTR #include <sys/wait.h>
Example
#include <sys/wait.h>
#include <unistd.h>
#include <stdio.h>

int main() {
    pid_t pid = fork();
    if (pid == 0) return 42; /* child exits with 42 */

    int status;
    waitpid(pid, &status, 0);

    if (WIFEXITED(status))
        printf("Child exited: %d\n", WEXITSTATUS(status));
}
🔗
pipe
int pipe(int pipefd[2])

Creates a unidirectional data channel. pipefd[0] is the read end, pipefd[1] is the write end. Data written to the write end can be read from the read end. Typically used between parent and child processes created with fork().

returns 0 on success errno: EMFILE, ENFILE #include <unistd.h>
Example
#include <unistd.h>
#include <stdio.h>
#include <string.h>

int main() {
    int  fd[2];
    char buf[64];

    pipe(fd);

    if (fork() == 0) {
        close(fd[0]);                          /* child writes */
        write(fd[1], "hello", 5);
        close(fd[1]);
    } else {
        close(fd[1]);                          /* parent reads */
        ssize_t n = read(fd[0], buf, sizeof(buf));
        printf("Got: %.*s\n", (int)n, buf);
    }
}
📋
dup
int dup(int oldfd)

Duplicates an existing file descriptor, returning the lowest-numbered available file descriptor that refers to the same open file description. Both descriptors share file offset and flags.

returns new fd or -1 errno: EBADF, EMFILE #include <unistd.h>
Example
#include <unistd.h>
#include <fcntl.h>
#include <stdio.h>

int main() {
    int fd  = open("out.txt", O_WRONLY | O_CREAT | O_TRUNC, 0644);
    int fd2 = dup(fd);         /* fd2 mirrors fd */

    write(fd,  "from fd\n",  8);
    write(fd2, "from fd2\n", 9); /* both write to same file */

    close(fd); close(fd2);
}
📋
dup2
int dup2(int oldfd, int newfd)

Like dup(), but uses exactly newfd as the new descriptor. If newfd is already open, it is silently closed first. Commonly used to redirect stdin/stdout/stderr to files or pipes.

returns newfd or -1 errno: EBADF, EINTR #include <unistd.h>
Example — redirect stdout to file
#include <unistd.h>
#include <fcntl.h>
#include <stdio.h>

int main() {
    int fd = open("output.txt", O_WRONLY | O_CREAT | O_TRUNC, 0644);

    dup2(fd, STDOUT_FILENO);  /* stdout now goes to file */
    close(fd);

    printf("This goes to output.txt\n");
}

Signals & Error Handling

Signals
⚡
kill
int kill(pid_t pid, int sig)

Sends signal sig to a process or process group identified by pid. If pid is 0, sends to all processes in the current group. If sig is 0, no signal is sent but error checking is performed (useful to test if a process exists).

returns 0 or -1 errno: EPERM, ESRCH #include <signal.h>
Example
#include <signal.h>
#include <stdio.h>

int main() {
    pid_t pid = 1234; /* target PID */

    if (kill(pid, 0) == 0)
        printf("Process %d exists\n", pid);

    kill(pid, SIGTERM); /* graceful terminate */
    /* kill(pid, SIGKILL); force kill */
}
📡
signal
sighandler_t signal(int signum, sighandler_t handler)

Installs a custom handler for a signal. Pass SIG_DFL to restore default behavior or SIG_IGN to ignore. Note: sigaction() is preferred on modern systems for reliable behavior across platforms.

previous handler or SIG_ERR errno: EINVAL #include <signal.h>
Example
#include <signal.h>
#include <stdio.h>

void handler(int sig) {
    printf("Caught signal %d\n", sig);
}

int main() {
    signal(SIGINT, handler);  /* Ctrl+C handler */
    signal(SIGTERM, SIG_IGN); /* ignore SIGTERM */

    while (1) pause();        /* wait for signals */
}
🔍
strerror
char * strerror(int errnum)

Converts an error number (typically from errno) to a human-readable string. The returned pointer points to a static buffer that may be overwritten by subsequent calls. Not thread-safe — use strerror_r() in multi-threaded code.

pointer to error string #include <string.h>
Example
#include <string.h>
#include <errno.h>
#include <stdio.h>
#include <fcntl.h>

int main() {
    int fd = open("/no/such/file", O_RDONLY);
    if (fd < 0)
        fprintf(stderr, "Error %d: %s\n", errno, strerror(errno));
    /* Output: Error 2: No such file or directory */
}
🚨
errno
extern int errno

A thread-local integer variable set by system calls and library functions to indicate what error occurred. It is only meaningful immediately after a failed call — successful calls may set it to any value. Always check return value first, then inspect errno.

#include <errno.h>
Example
#include <errno.h>
#include <string.h>
#include <stdio.h>
#include <unistd.h>

int main() {
    errno = 0;  /* reset before call */

    if (access("/root/secret", R_OK) == -1) {
        switch (errno) {
            case EACCES: puts("Permission denied"); break;
            case ENOENT: puts("File not found");    break;
            default:     perror("access");
        }
    }
}
🔎
gai_strerror
const char * gai_strerror(int ecode)

Returns a human-readable string for error codes returned by getaddrinfo() and getnameinfo(). These errors have their own namespace (e.g. EAI_NONAME, EAI_FAIL) and cannot be decoded with strerror().

pointer to string #include <netdb.h>
Example
#include <netdb.h>
#include <stdio.h>

int main() {
    struct addrinfo hints = {0}, *res;
    hints.ai_family   = AF_INET;
    hints.ai_socktype = SOCK_STREAM;

    int rc = getaddrinfo("invalid..host", "80", &hints, &res);
    if (rc != 0)
        fprintf(stderr, "getaddrinfo: %s\n", gai_strerror(rc));
}

Network & Sockets

Network
🌐
socket
int socket(int domain, int type, int protocol)

Creates an endpoint for network communication. Returns a file descriptor. domain: AF_INET (IPv4) or AF_INET6 (IPv6). type: SOCK_STREAM (TCP) or SOCK_DGRAM (UDP). Protocol is usually 0 to auto-select.

fd or -1 errno: EACCES, EMFILE, EPROTONOSUPPORT #include <sys/socket.h>
Example
#include <sys/socket.h>
#include <stdio.h>

int main() {
    /* TCP/IPv4 socket */
    int tcp_fd = socket(AF_INET, SOCK_STREAM, 0);

    /* UDP/IPv4 socket */
    int udp_fd = socket(AF_INET, SOCK_DGRAM, 0);

    if (tcp_fd < 0) perror("socket");

    printf("TCP fd=%d  UDP fd=%d\n", tcp_fd, udp_fd);
}
🔌
socketpair
int socketpair(int domain, int type, int protocol, int sv[2])

Creates a pair of connected sockets. Each end can be used for both reading and writing (full-duplex), unlike pipes. Primarily used with AF_UNIX for IPC between related processes. Both sockets are pre-connected to each other.

0 or -1 errno: EAFNOSUPPORT, EMFILE #include <sys/socket.h>
Example
#include <sys/socket.h>
#include <unistd.h>
#include <stdio.h>

int main() {
    int  sv[2];
    char buf[32];

    socketpair(AF_UNIX, SOCK_STREAM, 0, sv);

    if (fork() == 0) {
        close(sv[0]);
        write(sv[1], "ping", 4);
    } else {
        close(sv[1]);
        int n = read(sv[0], buf, sizeof(buf));
        printf("Got: %.*s\n", n, buf); /* "ping" */
    }
}
📌
bind
int bind(int sockfd, const struct sockaddr *addr, socklen_t addrlen)

Assigns a local address (IP + port) to a socket. Server sockets must be bound before listening. Use INADDR_ANY to bind on all interfaces. Use SO_REUSEADDR to avoid "Address already in use" on restart.

0 or -1 errno: EADDRINUSE, EACCES #include <sys/socket.h>
Example
#include <sys/socket.h>
#include <netinet/in.h>

int main() {
    int fd = socket(AF_INET, SOCK_STREAM, 0);

    struct sockaddr_in addr = {
        .sin_family      = AF_INET,
        .sin_port        = htons(8080),
        .sin_addr.s_addr = INADDR_ANY
    };

    if (bind(fd, (struct sockaddr *)&addr, sizeof(addr)) < 0)
        perror("bind");
}
👂
listen
int listen(int sockfd, int backlog)

Marks a socket as passive — ready to accept incoming connections. backlog specifies the maximum length of the queue of pending connections. Typical value is 128 (or SOMAXCONN). Must be called after bind().

0 or -1 errno: EADDRINUSE, ENOTSOCK #include <sys/socket.h>
Example — server setup
int fd = socket(AF_INET, SOCK_STREAM, 0);
/* ... bind() ... */

if (listen(fd, SOMAXCONN) < 0) {
    perror("listen");
    return 1;
}
printf("Listening on port 8080...\n");
🤝
accept
int accept(int sockfd, struct sockaddr *addr, socklen_t *addrlen)

Extracts the first pending connection from the listening socket's queue and creates a new connected socket. The original socket stays open for more connections. addr is filled with the remote peer's address.

new connected fd or -1 errno: EAGAIN, EINTR, EMFILE #include <sys/socket.h>
Example
struct sockaddr_in client;
socklen_t clen = sizeof(client);

while (1) {
    int cfd = accept(srv_fd, (struct sockaddr *)&client, &clen);
    if (cfd < 0) { perror("accept"); break; }

    printf("Client connected: %s\n",
           inet_ntoa(client.sin_addr));

    handle_client(cfd);  /* your logic here */
    close(cfd);
}
🔗
connect
int connect(int sockfd, const struct sockaddr *addr, socklen_t addrlen)

Initiates a connection from a client socket to a server at the given address. For TCP, this completes the three-way handshake. On non-blocking sockets returns EINPROGRESS immediately; use select/poll to wait.

0 or -1 errno: ECONNREFUSED, ETIMEDOUT, EINPROGRESS #include <sys/socket.h>
Example — TCP client
#include <sys/socket.h>
#include <netinet/in.h>
#include <arpa/inet.h>

int main() {
    int fd = socket(AF_INET, SOCK_STREAM, 0);

    struct sockaddr_in srv = {
        .sin_family      = AF_INET,
        .sin_port        = htons(80),
        .sin_addr.s_addr = inet_addr("93.184.216.34")
    };

    if (connect(fd, (struct sockaddr *)&srv, sizeof(srv)) < 0)
        perror("connect");
}
📤
send
ssize_t send(int sockfd, const void *buf, size_t len, int flags)

Transmits data over a connected socket. Unlike write(), allows flags such as MSG_NOSIGNAL (suppress SIGPIPE), MSG_DONTWAIT (non-blocking). May return fewer bytes than requested — always check the return value and loop.

bytes sent or -1 errno: EAGAIN, ENOTCONN, EPIPE #include <sys/socket.h>
Example
const char *msg = "GET / HTTP/1.0\r\n\r\n";
size_t total = 0, len = strlen(msg);

while (total < len) {
    ssize_t n = send(fd, msg + total,
                       len - total, MSG_NOSIGNAL);
    if (n <= 0) { perror("send"); break; }
    total += n;
}
📥
recv
ssize_t recv(int sockfd, void *buf, size_t len, int flags)

Receives data from a connected socket. Returns 0 when the peer has closed the connection (EOF). Use MSG_PEEK to inspect data without consuming it, or MSG_WAITALL to wait until the full buffer is filled.

bytes received, 0=EOF, -1=error errno: EAGAIN, EINTR, ENOTCONN #include <sys/socket.h>
Example
char buf[4096];
ssize_t n;

while ((n = recv(fd, buf, sizeof(buf) - 1, 0)) > 0) {
    buf[n] = '\0';
    printf("%s", buf);
}

if (n == 0)       puts("Connection closed by peer");
else if (n < 0) perror("recv");
🔄
htons / htonl / ntohs / ntohl
uint16_t htons(uint16_t)  |  uint32_t htonl(uint32_t)

Convert between host byte order and network byte order (big-endian). Network protocols require big-endian. host-to-network for short (16-bit) and long (32-bit); reverse: ntohs/ntohl. Always use these for port numbers and IP addresses.

converted integer value #include <arpa/inet.h>
Example
#include <arpa/inet.h>
#include <stdio.h>

int main() {
    uint16_t port    = 8080;
    uint32_t ip_host = 0x7F000001; /* 127.0.0.1 */

    uint16_t net_port = htons(port);
    uint32_t net_ip   = htonl(ip_host);

    printf("Port in network order: 0x%04X\n", net_port);

    /* Reverse: network → host */
    printf("Port back: %d\n", ntohs(net_port));
    printf("IP back:   0x%08X\n", ntohl(net_ip));
}
🧭
getaddrinfo
int getaddrinfo(const char *node, const char *service, const struct addrinfo *hints, struct addrinfo **res)

Resolves a hostname and service to a list of struct addrinfo structures suitable for use with socket() and connect()/bind(). Preferred over the legacy gethostbyname(). Handles both IPv4 and IPv6 automatically.

0 or EAI_* error code #include <netdb.h>
Example
#include <netdb.h>
#include <stdio.h>

int main() {
    struct addrinfo hints = {
        .ai_family   = AF_UNSPEC,    /* IPv4 or IPv6 */
        .ai_socktype = SOCK_STREAM,
    };
    struct addrinfo *res;

    int rc = getaddrinfo("example.com", "80", &hints, &res);
    if (rc != 0) {
        fprintf(stderr, "%s\n", gai_strerror(rc));
        return 1;
    }
    /* use res to create socket & connect */
    freeaddrinfo(res);
}
🗑️
freeaddrinfo
void freeaddrinfo(struct addrinfo *res)

Frees the linked list of addrinfo structures returned by getaddrinfo(). Must always be called after you're done with the results to avoid memory leaks. Frees the entire chain.

void #include <netdb.h>
Example
struct addrinfo *res = NULL;
getaddrinfo("example.com", "443", &hints, &res);

/* iterate through res linked list */
for (struct addrinfo *p = res; p != NULL; p = p->ai_next) {
    /* try each address */
}

freeaddrinfo(res);  /* always free after use */
res = NULL;
⚙
setsockopt
int setsockopt(int sockfd, int level, int optname, const void *optval, socklen_t optlen)

Sets socket options. Common uses: SO_REUSEADDR (allow port reuse after crash), SO_KEEPALIVE (detect dead connections), SO_RCVBUF/SO_SNDBUF (buffer sizes), TCP_NODELAY (disable Nagle's algorithm).

0 or -1 errno: EBADF, ENOPROTOOPT #include <sys/socket.h>
Example
int fd  = socket(AF_INET, SOCK_STREAM, 0);
int yes = 1;

/* Allow port reuse on restart */
setsockopt(fd, SOL_SOCKET, SO_REUSEADDR, &yes, sizeof(yes));

/* Disable Nagle's algorithm (low latency) */
setsockopt(fd, IPPROTO_TCP, TCP_NODELAY, &yes, sizeof(yes));

/* Set receive buffer to 64KB */
int bufsz = 65536;
setsockopt(fd, SOL_SOCKET, SO_RCVBUF, &bufsz, sizeof(bufsz));
📋
getsockname
int getsockname(int sockfd, struct sockaddr *addr, socklen_t *addrlen)

Retrieves the local address that the socket is bound to. Useful when the OS assigns an ephemeral port (when you bind to port 0) and you need to know which port was actually assigned.

0 or -1 errno: EBADF, ENOBUFS #include <sys/socket.h>
Example — find assigned port
struct sockaddr_in addr = {
    .sin_family      = AF_INET,
    .sin_port        = 0,          /* OS picks port */
    .sin_addr.s_addr = INADDR_ANY
};
bind(fd, (struct sockaddr *)&addr, sizeof(addr));

socklen_t len = sizeof(addr);
getsockname(fd, (struct sockaddr *)&addr, &len);
printf("Bound to port: %d\n", ntohs(addr.sin_port));
📖
getprotobyname
struct protoent * getprotobyname(const char *name)

Looks up a protocol by its name (e.g., "tcp", "udp", "icmp") and returns a struct protoent containing the protocol number. Reads from /etc/protocols. Returns NULL if not found.

struct protoent * or NULL #include <netdb.h>
Example
#include <netdb.h>
#include <stdio.h>
#include <sys/socket.h>

int main() {
    struct protoent *pe = getprotobyname("tcp");
    if (!pe) { puts("unknown protocol"); return 1; }

    printf("Protocol: %s  Number: %d\n",
           pe->p_name, pe->p_proto);

    /* Use protocol number explicitly */
    int fd = socket(AF_INET, SOCK_STREAM, pe->p_proto);
}

I/O Multiplexing

I/O
📻
select
int select(int nfds, fd_set *readfds, fd_set *writefds, fd_set *exceptfds, struct timeval *timeout)

Monitors multiple file descriptors simultaneously, waiting until at least one is ready for I/O. Limited to FD_SETSIZE (1024) descriptors. Portable but older — prefer poll or epoll for new code. Modifies the fd_sets in place.

ready fds count, 0=timeout, -1=error errno: EBADF, EINTR, EINVAL #include <sys/select.h>
Example
#include <sys/select.h>
#include <stdio.h>

int main() {
    fd_set readfds;
    struct timeval tv = { .tv_sec = 5 }; /* 5s timeout */

    FD_ZERO(&readfds);
    FD_SET(STDIN_FILENO, &readfds);
    FD_SET(server_fd,    &readfds);

    int ret = select(server_fd + 1, &readfds, NULL, NULL, &tv);
    if (ret > 0 && FD_ISSET(server_fd, &readfds))
        puts("Server socket is readable");
}
📊
poll
int poll(struct pollfd *fds, nfds_t nfds, int timeout)

Modern replacement for select(). No hard limit on file descriptors. Each entry in fds specifies events to watch (POLLIN, POLLOUT). After the call, revents is populated with what actually happened. timeout is in milliseconds (-1 = infinite).

ready count, 0=timeout, -1=error errno: EFAULT, EINTR #include <poll.h>
Example
#include <poll.h>
#include <stdio.h>

int main() {
    struct pollfd fds[2] = {
        { .fd = server_fd, .events = POLLIN },
        { .fd = client_fd, .events = POLLIN | POLLOUT }
    };

    int ret = poll(fds, 2, 3000); /* 3 second timeout */
    if (ret > 0) {
        if (fds[0].revents & POLLIN) puts("server: incoming conn");
        if (fds[1].revents & POLLIN) puts("client: data ready");
    }
}
⚡
epoll (Linux)
int epoll_create | epoll_ctl | epoll_wait

High-performance Linux I/O event notification. Scales to thousands of connections without degrading (unlike select/poll). epoll_create1() creates an epoll instance. epoll_ctl() adds/removes/modifies watched fds. epoll_wait() blocks until events arrive. Supports edge-triggered (EPOLLET) mode.

epoll_wait: event count or -1 errno: EBADF, EINVAL, ENOSPC #include <sys/epoll.h>
Example — event loop
#include <sys/epoll.h>
#define MAX_EVENTS 64

int epfd = epoll_create1(0);

struct epoll_event ev = {
    .events  = EPOLLIN | EPOLLET,  /* edge-triggered */
    .data.fd = server_fd
};
epoll_ctl(epfd, EPOLL_CTL_ADD, server_fd, &ev);

struct epoll_event events[MAX_EVENTS];
while (1) {
    int n = epoll_wait(epfd, events, MAX_EVENTS, -1);
    for (int i = 0; i < n; i++) {
        if (events[i].data.fd == server_fd)
            handle_new_connection(server_fd);
        else
            handle_client(events[i].data.fd);
    }
}
🍎
kqueue (BSD/macOS)
int kqueue()  |  int kevent(int kq, ...)

BSD/macOS equivalent of epoll. kqueue() creates the kernel event queue. kevent() both registers interest (changelist) and retrieves fired events (eventlist) in a single call. More versatile than epoll — watches files, sockets, processes, signals, and timers.

event count or -1 errno: EBADF, ENOMEM #include <sys/event.h>
Example
#include <sys/event.h>
#include <stdio.h>

int main() {
    int kq = kqueue();

    struct kevent change;
    EV_SET(&change, server_fd, EVFILT_READ, EV_ADD, 0, 0, NULL);
    kevent(kq, &change, 1, NULL, 0, NULL); /* register */

    struct kevent events[16];
    while (1) {
        int n = kevent(kq, NULL, 0, events, 16, NULL);
        for (int i = 0; i < n; i++)
            handle((int)events[i].ident);
    }
}

File System & I/O

File System
📂
open
int open(const char *path, int flags, ... mode_t mode)

Opens a file and returns a file descriptor. Flags: O_RDONLY, O_WRONLY, O_RDWR, O_CREAT, O_TRUNC, O_APPEND, O_NONBLOCK. The mode argument (octal permissions) is required only with O_CREAT.

fd ≥ 0 or -1 errno: ENOENT, EACCES, EMFILE #include <fcntl.h>
Example
#include <fcntl.h>
#include <unistd.h>

int main() {
    /* Read existing file */
    int rfd = open("input.txt", O_RDONLY);

    /* Create/truncate for writing, perms 0644 */
    int wfd = open("output.txt",
                   O_WRONLY | O_CREAT | O_TRUNC,
                   0644);

    /* Append mode */
    int afd = open("log.txt", O_WRONLY | O_APPEND, 0644);

    close(rfd); close(wfd); close(afd);
}
❌
close
int close(int fd)

Closes an open file descriptor, releasing kernel resources. The fd is freed and may be reused. Flushed data may still be in the kernel buffer. Important: on EINTR, the fd state is unspecified on Linux — do not retry close() on EINTR.

0 or -1 errno: EBADF, EINTR, EIO #include <unistd.h>
Example
#include <unistd.h>
#include <fcntl.h>
#include <stdio.h>

int main() {
    int fd = open("file.txt", O_RDONLY);
    if (fd < 0) { perror("open"); return 1; }

    /* ... use fd ... */

    if (close(fd) < 0)
        perror("close"); /* EIO → data may be lost */
}
📖
read
ssize_t read(int fd, void *buf, size_t count)

Reads up to count bytes from fd into buf. Returns 0 at EOF. May return fewer bytes than requested — always loop until 0 or error. Advances the file offset. Works on files, pipes, sockets, and devices.

bytes read, 0=EOF, -1=error errno: EAGAIN, EINTR, EIO #include <unistd.h>
Example
#include <unistd.h>
#include <fcntl.h>
#include <stdio.h>

int main() {
    int     fd = open("data.bin", O_RDONLY);
    char    buf[512];
    ssize_t n;

    while ((n = read(fd, buf, sizeof(buf))) > 0)
        write(STDOUT_FILENO, buf, n);

    if (n < 0) perror("read");
    close(fd);
}
✏️
write
ssize_t write(int fd, const void *buf, size_t count)

Writes up to count bytes from buf to fd. May write fewer bytes than requested (partial write) — loop until all data is written. For pipes, writes ≤ PIPE_BUF bytes are guaranteed atomic.

bytes written or -1 errno: ENOSPC, EPIPE, EINTR #include <unistd.h>
Example — safe write loop
ssize_t write_all(int fd, const void *buf, size_t n) {
    size_t sent = 0;
    while (sent < n) {
        ssize_t r = write(fd, (char*)buf + sent, n - sent);
        if (r <= 0) return r;
        sent += r;
    }
    return sent;
}

stat
int stat(const char *pathname, struct stat *statbuf)

Retrieves file metadata: size, permissions, owner, timestamps, inode number, and type. lstat() does not follow symbolic links. fstat() operates on an open file descriptor. Check file type with macros: S_ISREG, S_ISDIR, S_ISLNK.

0 or -1 errno: ENOENT, EACCES, ELOOP #include <sys/stat.h>
Example
#include <sys/stat.h>
#include <stdio.h>

int main() {
    struct stat sb;

    if (stat("/etc/passwd", &sb) < 0) {
        perror("stat"); return 1;
    }

    printf("Size:  %lld bytes\n", (long long)sb.st_size);
    printf("Inode: %lu\n", (unsigned long)sb.st_ino);
    printf("Mode:  %o\n", sb.st_mode);
    printf("Is dir: %s\n", S_ISDIR(sb.st_mode) ? "yes" : "no");
}
🔐
access
int access(const char *pathname, int mode)

Checks whether the calling process can access a file. mode: F_OK (exists), R_OK (readable), W_OK (writable), X_OK (executable). Uses the real UID/GID, not effective — making it unsuitable for setuid programs.

0 if allowed, -1 if not errno: EACCES, ENOENT, ELOOP #include <unistd.h>
Example
#include <unistd.h>
#include <stdio.h>

int main() {
    if (access("/etc/shadow", F_OK) == 0)
        puts("File exists");

    if (access("/etc/shadow", R_OK) != 0)
        puts("Not readable (permission denied)");

    if (access("/usr/bin/ls", X_OK) == 0)
        puts("/usr/bin/ls is executable");
}
🔧
fcntl
int fcntl(int fd, int cmd, ... arg)

Swiss-army knife for file descriptor control. Common uses: F_GETFL/F_SETFL (get/set flags like O_NONBLOCK), F_DUPFD (duplicate fd), F_GETFD/F_SETFD (close-on-exec flag), file locking with F_SETLK.

depends on cmd, -1 on error errno: EBADF, EINVAL, EMFILE #include <fcntl.h>
Example — set non-blocking
#include <fcntl.h>

void set_nonblocking(int fd) {
    int flags = fcntl(fd, F_GETFL, 0);
    if (flags < 0) { perror("fcntl F_GETFL"); return; }
    fcntl(fd, F_SETFL, flags | O_NONBLOCK);
}

void set_cloexec(int fd) {
    /* fd closes automatically on exec() */
    fcntl(fd, F_SETFD, FD_CLOEXEC);
}
📁
chdir
int chdir(const char *path)

Changes the current working directory of the calling process. All subsequent relative paths will resolve from this new location. Note that this only affects the current process — it does not affect the parent shell or other processes.

0 or -1 errno: ENOENT, EACCES, ENOTDIR #include <unistd.h>
Example
#include <unistd.h>
#include <stdio.h>

int main() {
    char cwd[256];
    getcwd(cwd, sizeof(cwd));
    printf("Before: %s\n", cwd);

    if (chdir("/tmp") < 0) {
        perror("chdir"); return 1;
    }
    getcwd(cwd, sizeof(cwd));
    printf("After:  %s\n", cwd);
}
📂
opendir
DIR * opendir(const char *name)

Opens a directory stream for reading its entries. Returns a DIR* pointer used by readdir() and closedir(). The directory stream is positioned at the first entry. Use fdopendir() to open from an existing file descriptor.

DIR* or NULL errno: ENOENT, EACCES, ENOTDIR #include <dirent.h>
Example
#include <dirent.h>
#include <stdio.h>

int main() {
    DIR *dir = opendir("/tmp");
    if (!dir) { perror("opendir"); return 1; }

    struct dirent *entry;
    while ((entry = readdir(dir)) != NULL)
        printf("%s\n", entry->d_name);

    closedir(dir);
}
📋
readdir
struct dirent * readdir(DIR *dirp)

Reads the next entry from a directory stream. Returns a pointer to a struct dirent containing d_name (filename) and d_type (DT_REG, DT_DIR, etc.). Returns NULL at end-of-directory or on error. Not thread-safe — use readdir_r() for threads.

struct dirent* or NULL #include <dirent.h>
Example — filter files by type
struct dirent *e;
DIR *dir = opendir(".");

while ((e = readdir(dir)) != NULL) {
    if (e->d_type == DT_REG)
        printf("FILE: %s\n", e->d_name);
    else if (e->d_type == DT_DIR
             && strcmp(e->d_name, ".")  != 0
             && strcmp(e->d_name, "..") != 0)
        printf(" DIR: %s\n", e->d_name);
}
closedir(dir);
🚪
closedir
int closedir(DIR *dirp)

Closes a directory stream opened by opendir(), freeing associated resources and the underlying file descriptor. Always pair with opendir() to prevent resource leaks. The pointer is invalid after this call.

0 or -1 errno: EBADF #include <dirent.h>
Example — always close in error paths
void list_dir(const char *path) {
    DIR *dir = opendir(path);
    if (!dir) { perror("opendir"); return; }

    struct dirent *e;
    while ((e = readdir(dir)))
        puts(e->d_name);

    if (closedir(dir) < 0)
        perror("closedir");
}
